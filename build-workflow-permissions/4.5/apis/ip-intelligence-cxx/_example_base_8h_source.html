34// Times in milliseconds. Windows uses the performance counter, as the tick
35// count only moves every 15 or 16 milliseconds, which is as long as some
36// timed runs take.
37#ifdef _MSC_VER
38#define TIMER_CREATE LARGE_INTEGER start, end, frequency
39#define TIMER_START (QueryPerformanceFrequency(&frequency), \
40QueryPerformanceCounter(&start))
41#define TIMER_END QueryPerformanceCounter(&end)
42#define TIMER_ELAPSED ((double)(end.QuadPart - start.QuadPart) * 1000.0 / \
43(double)frequency.QuadPart)
44#else
45#define TIMER_CREATE struct timespec start, end
46#define TIMER_START clock_gettime(CLOCK_MONOTONIC, &start)
47#define TIMER_END clock_gettime(CLOCK_MONOTONIC, &end)
48#define TIMER_ELAPSED (((end.tv_sec - start.tv_sec) + \
49(end.tv_nsec - start.tv_nsec) / 1.0e9) * 1000.0)
50#endif
